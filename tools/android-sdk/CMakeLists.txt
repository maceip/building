# Builds the pinned AOSP SDK helpers using lzhiyong/android-sdk-tools target definitions.
cmake_minimum_required(VERSION 3.20)
project(building_sdk_tools C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
add_compile_options("-ffile-prefix-map=${PROJECT_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_ANDROID_NDK}=/ndk")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fPIC -Wno-attributes")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC -Wno-attributes -D__ANDROID_UNAVAILABLE_SYMBOLS_ARE_WEAK__")
add_compile_definitions(_FILE_OFFSET_BITS=64 ART_STACK_OVERFLOW_GAP_arm=16384 ART_STACK_OVERFLOW_GAP_arm64=16384 ART_STACK_OVERFLOW_GAP_riscv64=16384 ART_STACK_OVERFLOW_GAP_x86=16384 ART_STACK_OVERFLOW_GAP_x86_64=20480 ART_FRAME_SIZE_LIMIT=10000 ART_BASE_ADDRESS=0x70000000)
set(SRC ${PROJECT_SOURCE_DIR}/src)
set(BUILD_SHARED_LIBS OFF)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_WITH_POST_BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(src/boringssl EXCLUDE_FROM_ALL)
add_subdirectory(src/fmtlib EXCLUDE_FROM_ALL)
add_subdirectory(src/pcre EXCLUDE_FROM_ALL)
add_subdirectory(src/libpng EXCLUDE_FROM_ALL)
add_subdirectory(src/zopfli EXCLUDE_FROM_ALL)
add_subdirectory(src/jsoncpp EXCLUDE_FROM_ALL)
add_subdirectory(src/expat/expat EXCLUDE_FROM_ALL)
add_subdirectory(src/tinyxml2 EXCLUDE_FROM_ALL)
foreach(lib base log cutils utils androidfw incfs selinux sepol ziparchive buildversion packagelistparser processgroup)
 include(lib/lib${lib}.cmake)
endforeach()
set(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/bin)
add_library(sdk-exec-prefix SHARED exec-prefix.c)
target_link_libraries(sdk-exec-prefix PRIVATE dl)
target_link_options(sdk-exec-prefix PRIVATE "-Wl,-z,max-page-size=16384,-z,common-page-size=16384")
foreach(tool aapt dexdump split-select zipalign)
 include(build-tools/${tool}.cmake)
endforeach()
# Android API 26 has no reallocarray; libsepol supplies its checked fallback.
get_target_property(sepol_defines libsepol COMPILE_DEFINITIONS)
list(REMOVE_ITEM sepol_defines HAVE_REALLOCARRAY -DHAVE_REALLOCARRAY)
set_target_properties(libsepol PROPERTIES COMPILE_DEFINITIONS "${sepol_defines}")
